    50	Texture Image::AllocateAndSetupDataFromMemory(int width, int height,
    51	                                              GLenum format, const void *data) {
    52	  if (!data) {
    53	    return {};
    54	  }
    55	
    56	  Texture result;
    57	  result.Width = width;
    58	  result.Height = height;
    59	
    60	  glCreateTextures(GL_TEXTURE_2D, 1, &result.Handle);
    61	
    62	  glTextureStorage2D(result.Handle, 1, (format == GL_RGBA ? GL_RGBA8 : GL_RGB8),
    63	                     width, height);
    64	
    65	  glTextureSubImage2D(result.Handle, 0, 0, 0, width, height, format,
    66	                      GL_UNSIGNED_BYTE, data);
    67	
    68	  glTextureParameteri(result.Handle, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
    69	  glTextureParameteri(result.Handle, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
    70	
    71	  glTextureParameteri(result.Handle, GL_TEXTURE_WRAP_S, GL_REPEAT);
    72	  glTextureParameteri(result.Handle, GL_TEXTURE_WRAP_T, GL_REPEAT);
    73	
    74	  glGenerateTextureMipmap(result.Handle);
    75	
    76	  return result;

   105	bool Image::AttachTextureToFramebuffer(Framebuffer &framebuffer,
   106	                                       const Texture texture) {
   107	  glNamedFramebufferTexture(framebuffer.Handle, GL_COLOR_ATTACHMENT0,
   108	                            texture.Handle, 0);
   109	
   110	  if (glCheckFramebufferStatus(GL_FRAMEBUFFER) != GL_FRAMEBUFFER_COMPLETE) {
   111	    std::cerr << "Framebuffer is not complete!" << std::endl;
   112	    return false;
   113	  }
   114	
   115	  framebuffer.ColorAttachment = texture;
   116	  return true;
   117	}

   119	void Image::BlitFramebufferToSwapchain(const Framebuffer framebuffer) {
   120	  glBindFramebuffer(GL_READ_FRAMEBUFFER, framebuffer.Handle);
   121	  glBindFramebuffer(GL_DRAW_FRAMEBUFFER, 0); // swapchain
   122	
   123	  glBlitFramebuffer(0, 0, framebuffer.ColorAttachment.Width,
   124	                    framebuffer.ColorAttachment.Height, // Source rect
   125	                    0, 0, framebuffer.ColorAttachment.Width,
   126	                    framebuffer.ColorAttachment.Height, // Destination rect
   127	                    GL_COLOR_BUFFER_BIT, GL_NEAREST);
   128	}
